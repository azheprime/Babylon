                        const vertex = `
    precision highp float;
    attribute vec3 position;
    uniform mat4 world;
    uniform mat4 viewProjection;
    varying vec4 vScreenUV;

    vec4 ComputeScreenPos(vec4 pos){
        vec4 o = pos*0.5;
        o.xy = vec2(o.x,o.y*1.) + o.w;
        o.zw = pos.zw;
        return o;
    }

    void main(void){
        gl_Position = viewProjection * world * vec4(position,1.);
        vScreenUV = ComputeScreenPos(gl_Position);
        
    }
`

const fragment = `
    precision highp float;


    uniform sampler2D depthMap;

    uniform vec2 cameraData;

    varying vec4 vScreenUV;



    void main(void){


        vec2 screenUV = vScreenUV.xy/vScreenUV.w;

        float dSampler = texture2D(depthMap,screenUV).x* (cameraData.y + cameraData.x) - cameraData.x;


        if(dSampler>50.){gl_FragColor = vec4(1.,0.,0.,1.);}
        else{gl_FragColor =vec4(1.,1.,0.,1.);}
        
//gl_FragColor = vec4(1.,1.,0.,1.);



    }

`

















                        BABYLON.Effect.ShadersStore['CustomTranVertexShader'] = vertex;

                            attributes: ['position'],
                            uniforms: ['viewProjection', 'world', 'cameraData'],
                            samplers: ['depthMap']
                        })


                        //-----------------------------------------DefaultMaterial--------------------------------------------------
