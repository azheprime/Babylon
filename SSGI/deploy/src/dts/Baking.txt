vec3 createalight_zero (vec2 coord ,float i,float j)
{
//coord对于不同部分，似乎是不公平的
vec2 prerand1=coord*i;
vec2 prerand2=coord*0.5353*i;

float r1= rand(prerand1);
float r2= rand(prerand2);
float r2active=abs(r2-j);//解决光线太贴近的自交问题；


    float theta = 2.0*3.1415926*r1;
    float fi = sqrt(r2);//注意此处适时改为r2active
     
    float x = fi * cos(theta);
    float y = fi * sin(theta);
    float z = sqrt(1.-r2active);

    vec3 light = vec3(x, y, z);


    return light;
}
//---------------------------------------camera原点的半球空间------------------------------------------------------







//--------------------------------------半球空间映射---------------------------------------------------------

vec3 createalight_rec (vec3 s,vec3 t,vec3 n,vec3 light)
{

    vec3 light_rec  =   light.x*s+light.y*t+light.z*n;

        float length_rec = sqrt(dot(light_rec, light_rec));

        light_rec=light_rec/length_rec;

    return light_rec;
}

function FiltTheLM(){

//-----------------------------------------------Filter base data-------------------------------------------------------------


var m_sigmaNormal =1/65536;
var m_sigmaNormal2 =0.5/65536;

var total_weight=0;
var D_normal=0;
var Kernelsize=10;
var Kernelsize2=4;
var subarrayfinalcolor1=0;
var subarrayfinalcolor2=0;
var subarrayfinalcolor3=0;

//-----------------------------------------------Filter base data-------------------------------------------------------------


//-----------------------------------------------Filter base data-------------------------------------------------------------

  for (var row = 0; row < lightmapheight ; row++) {
    for (var col = 0; col < lightmapwidth ; col++) {
      var i = (row * lightmapwidth + col) * 4;
          total_weight=0;

subarrayfinalcolor1=0;
subarrayfinalcolor2=0;
subarrayfinalcolor3=0;
 

  for (var row2 = row-Kernelsize; row2 <= row+Kernelsize ; row2++) {
    for (var col2 = col-Kernelsize; col2 <=col+Kernelsize ; col2++) {
      var i2 = (row2 * lightmapwidth + col2) * 4;


if(row2<lightmapheight  && i2<LMSize && i2>=0 ){   

          D_normal=m_sigmaNormal*(LMnormal[i]*LMnormal[i2]+LMnormal[i+1]*LMnormal[i2+1]+LMnormal[i+2]*LMnormal[i2+2]);

          subarrayfinalcolor1+=LMColor[i2]*D_normal;
          subarrayfinalcolor2 += LMColor[i2+1]*D_normal;
          subarrayfinalcolor3 += LMColor[i2+2]*D_normal;
          total_weight+=D_normal;




}

    }}


          FinalLMcolor[i]=Math.round( subarrayfinalcolor1/total_weight);
          FinalLMcolor[i+1] = Math.round( subarrayfinalcolor2/total_weight);
          FinalLMcolor[i+2] = Math.round( subarrayfinalcolor3/total_weight);
          FinalLMcolor[i+3] = 255;

    }}


//-----------------------------------------------Filter base data-------------------------------------------------------------








//-----------------------------------------------Filter base data-------------------------------------------------------------

  for (var row = 0; row < lightmapheight ; row++) {
    for (var col = 0; col < lightmapwidth ; col++) {
      var i = (row * lightmapwidth + col) * 4;
          total_weight=0;

subarrayfinalcolor1=0;
subarrayfinalcolor2=0;
subarrayfinalcolor3=0;
 

  for (var row2 = row-Kernelsize2; row2 <= row+Kernelsize2 ; row2++) {
    for (var col2 = col-Kernelsize2; col2 <=col+Kernelsize2 ; col2++) {
      var i2 = (row2 * lightmapwidth + col2) * 4;


if(row2<lightmapheight  && i2<LMSize && i2>=0 ){   

          D_normal=m_sigmaNormal2*(LMnormal[i]*LMnormal[i2]+LMnormal[i+1]*LMnormal[i2+1]+LMnormal[i+2]*LMnormal[i2+2]);

          subarrayfinalcolor1+=FinalLMcolor[i2]*D_normal;
          subarrayfinalcolor2 += FinalLMcolor[i2+1]*D_normal;
          subarrayfinalcolor3 += FinalLMcolor[i2+2]*D_normal;
          total_weight+=D_normal;




}

    }}


          FinalLMcolor[i]=Math.round( subarrayfinalcolor1/total_weight);
          FinalLMcolor[i+1] = Math.round( subarrayfinalcolor2/total_weight);
          FinalLMcolor[i+2] = Math.round( subarrayfinalcolor3/total_weight);
          FinalLMcolor[i+3] = 255;

    }}


//-----------------------------------------------Filter base data-------------------------------------------------------------




//------------------------------------------------compensate FinalLMcolor-------------------------------------------------------------
  for (var row = lightmapheight-2; row < lightmapheight ; row++) {
    for (var col = 0; col < lightmapwidth ; col++) {
      var i = (row * lightmapwidth + col) * 4;
      var j = ((row-5) * lightmapwidth + col) * 4;

            FinalLMcolor[i] = FinalLMcolor[j];
            FinalLMcolor[i+1] = FinalLMcolor[j+1];
            FinalLMcolor[i+2] = FinalLMcolor[j+2];
            FinalLMcolor[i+3] = FinalLMcolor[j+3];
    }}
//------------------------------------------------compensate FinalLMcolor-------------------------------------------------------------

}

        lastTime = currentTime; 
        count++;
    }
    Camera1.beta = BABYLON.Tools.ToRadians(90); 

    Camera1.alpha = BABYLON.Tools.ToRadians(currentAlpha); 

}

 Scene1.unregisterBeforeRender(CAMERAMOVING1);
